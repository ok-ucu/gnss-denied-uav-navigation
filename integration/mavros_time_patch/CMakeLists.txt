cmake_minimum_required(VERSION 3.16)
project(gnss_mavros_time_patch)
set(CMAKE_CXX_STANDARD 20)
find_package(ament_cmake REQUIRED)
find_package(mavros REQUIRED)
find_package(pluginlib REQUIRED)
find_package(diagnostic_updater REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(mavros_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(libmavconn REQUIRED)
find_package(eigen3_cmake_module REQUIRED)
find_package(Eigen3 REQUIRED)
foreach(dep tf2_ros tf2_eigen rclcpp_components geometry_msgs nav_msgs trajectory_msgs std_srvs)
  find_package(${dep} REQUIRED)
endforeach()
add_library(gnss_mavros_time_patch SHARED src/sys_time.cpp)
ament_target_dependencies(gnss_mavros_time_patch mavros pluginlib diagnostic_updater sensor_msgs mavros_msgs rclcpp libmavconn tf2_ros tf2_eigen rclcpp_components geometry_msgs nav_msgs trajectory_msgs std_srvs)
target_link_libraries(gnss_mavros_time_patch Eigen3::Eigen)
pluginlib_export_plugin_description_file(mavros plugins.xml)
install(TARGETS gnss_mavros_time_patch DESTINATION lib)
ament_package()
